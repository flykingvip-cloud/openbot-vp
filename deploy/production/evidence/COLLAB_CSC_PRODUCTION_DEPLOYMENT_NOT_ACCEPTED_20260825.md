## Appendix A — capture-chat-lifecycle-v3.cjs

sha256 `1c62b227932420219c06c8afc2747de01a4219ebb496520622a9a2720e07e9ce`. Token-only, read-only,
field-whitelisted, forbidden-key scan. Records the full chat.\* lifecycle and summarizes depth
events, mode (shadow vs router), reasons/classification, and model/tool counts. (The sha above is
the validated canonical tool; the docs formatter may reflow the embedded listing, which stays
functionally identical and passes `node -c`.)

```js
#!/usr/bin/env node
"use strict";
// Enhanced read-only shadow observation for a real-traffic window. Requires ONLY `ws`.
// Token-only connect (no device). Token read ONLY from OPENBOT_OPERATOR_TOKEN_FILE.
// Records the full chat.* lifecycle (server already redacts sensitive fields); this tool
// additionally whitelists recorded fields and scans for forbidden keys. Produces a summary
// that answers the real-event acceptance: depth events, mode=shadow (router off, incl.
// spaces), reasons/classification, and model/tool activity counts by runId.
const fs = require("node:fs");

const SRC = "OPENBOT_OPERATOR_TOKEN_FILE";
function argVal(name) {
  const i = process.argv.indexOf(name);
  return i >= 0 && i + 1 < process.argv.length ? process.argv[i + 1] : undefined;
}
if (process.argv.includes("--help") || process.argv.includes("-h")) {
  console.log(
    "usage: OPENBOT_OPERATOR_TOKEN_FILE=<file> node capture-chat-lifecycle-v3.cjs " +
      "[--gateway ws://127.0.0.1:18789] [--duration 300] [--output <path>]\n" +
      "Token is read ONLY from OPENBOT_OPERATOR_TOKEN_FILE (never argv). Read-only, token-only.",
  );
  process.exit(0);
}
const tokenFile = process.env.OPENBOT_OPERATOR_TOKEN_FILE;
const url = argVal("--gateway") || process.env.CAPTURE_GATEWAY || "ws://127.0.0.1:18789";
const duration = Number(argVal("--duration") || process.env.CAPTURE_DURATION || "300");
const output = argVal("--output") || process.env.CAPTURE_OUTPUT || "/tmp/chat-lifecycle.jsonl";

function fail(type) {
  console.error("AUTH_SOURCE=" + SRC + " ERROR_TYPE=" + type);
  process.exit(3);
}
let WebSocket;
try {
  WebSocket = require("ws");
} catch {
  console.error("AUTH_SOURCE=" + SRC + " ERROR_TYPE=ws_module_missing");
  process.exit(4);
}
if (!tokenFile) fail("env_unset");
let token = "";
try {
  token = fs.readFileSync(tokenFile, "utf8").trim();
} catch {
  fail("file_unreadable");
}
if (!token) fail("empty_token");

// Fields safe to persist from a chat.* lifecycle event's data (rationale/metrics only).
const ALLOW = new Set([
  "event",
  "wouldDepth",
  "routeStrength",
  "mode",
  "classifierVersion",
  "reasons",
  "routeMs",
  "sessionKeyHash",
  "flags",
]);
const FORBIDDEN = new Set([
  "text",
  "body",
  "content",
  "prompt",
  "message",
  "attachment",
  "attachments",
  "token",
  "secret",
  "password",
  "apikey",
  "authorization",
]);
const crypto = require("node:crypto");
void crypto;

const out = fs.createWriteStream(output, { flags: "a", mode: 0o600 });
let wsConnect = false;
let auth = false;
let subAck = false;
let nonDepth = false;
let leaks = 0;
let depth = 0;
let modelStarted = 0;
let toolEvents = 0;
let continuationSeen = false;
const modes = new Set();
const hist = Object.create(null);
const connectId = crypto.randomUUID();

function hasForbidden(o) {
  if (!o || typeof o !== "object") return false;
  for (const k of Object.keys(o)) if (FORBIDDEN.has(k.toLowerCase())) return true;
  return false;
}

const ws = new WebSocket(url);
ws.on("open", () => {
  wsConnect = true;
});
ws.on("error", (e) => {
  const m = String((e && e.message) || e).toLowerCase();
  console.error(
    "AUTH_SOURCE=" + SRC + " ERROR_TYPE=" + (/refused/.test(m) ? "connect_refused" : "ws_error"),
  );
});
ws.on("message", (raw) => {
  let msg;
  try {
    msg = JSON.parse(raw.toString());
  } catch {
    return;
  }
  if (msg.type === "event" && msg.event === "connect.challenge") {
    ws.send(
      JSON.stringify({
        type: "req",
        id: connectId,
        method: "connect",
        params: {
          minProtocol: 3,
          maxProtocol: 3,
          client: {
            id: "gateway-client",
            version: "1.0.0",
            platform: process.platform,
            mode: "backend",
          },
          caps: [],
          auth: { token },
          role: "operator",
          scopes: ["operator.admin"],
        },
      }),
    );
    return;
  }
  if (msg.type === "res" && msg.id === connectId) {
    if (msg.ok) {
      auth = true;
      wsConnect = true;
      subAck = true;
    } else {
      const em = String((msg.error && msg.error.message) || "").toLowerCase();
      console.error(
        "AUTH_SOURCE=" +
          SRC +
          " ERROR_TYPE=" +
          (/token/.test(em) ? "token_rejected" : "connect_rejected"),
      );
    }
    return;
  }
  if (msg.type !== "event") return;
  const payload = msg.payload || {};
  const data = (payload && payload.data) || {};
  const ev = String(data.event || "");
  const isChat = msg.event === "agent" && ev.indexOf("chat.") === 0;
  if (!isChat) {
    nonDepth = true;
    return;
  }
  if (hasForbidden(data) || hasForbidden(payload)) leaks++;
  hist[ev] = (hist[ev] || 0) + 1;
  if (ev !== "chat.depth_decided") nonDepth = true;
  if (ev === "chat.model_started") modelStarted++;
  if (ev === "chat.tool_policy_applied" || ev === "chat.tool_rejected") toolEvents++;
  if (ev === "chat.depth_decided") {
    depth++;
    if (data.mode) modes.add(String(data.mode));
    const reasons = Array.isArray(data.reasons) ? data.reasons.map(String) : [];
    if (reasons.some((r) => /continuation|unfinished/i.test(r))) continuationSeen = true;
    const rec = {};
    for (const k of Object.keys(data)) if (ALLOW.has(k)) rec[k] = data[k];
    rec.ts = payload.ts;
    rec.runId = payload.runId;
    rec.seq = payload.seq;
    out.write(JSON.stringify(rec) + "\n");
  }
});

setTimeout(() => {
  try {
    out.end();
  } catch {}
  const pf = (b) => (b ? "PASS" : "FAIL");
  const modeList = Array.from(modes).sort().join(",") || "none";
  const modeAllShadow = modes.size > 0 && Array.from(modes).every((m) => m === "shadow");
  const histStr = Object.keys(hist)
    .sort()
    .map((k) => k + "=" + hist[k])
    .join(" ");
  console.log(
    [
      "SMOKE_SUMMARY",
      "AUTH_SOURCE=" + SRC,
      "AUTH=" + pf(auth),
      "WS_CONNECT=" + pf(wsConnect),
      "SUBSCRIPTION_ACK=" + pf(subAck),
      "NON_DEPTH_EVENT_VISIBLE=" + (nonDepth ? "YES" : "NO"),
      "DEPTH_EVENTS=" + depth,
      "DEPTH_MODES=" + modeList,
      "MODE_ALL_SHADOW=" + (depth > 0 ? (modeAllShadow ? "YES" : "NO") : "NA"),
      "MODEL_STARTED=" + modelStarted,
      "TOOL_EVENTS=" + toolEvents,
      "CONTINUATION_SEEN=" + (continuationSeen ? "YES" : "NO"),
      "RAW_TEXT_LEAKS=" + leaks,
      "DEVICE_REGISTERED=NO",
      "LIVE_TOUCHED=NO",
      "FLAGS_CHANGED=NO",
      "EVENT_HISTOGRAM=" + (histStr || "none"),
    ].join("\n"),
  );
  try {
    ws.close();
  } catch {}
  // Non-zero only on link/redaction failure; DEPTH_EVENTS>0 and mode judgment are for the operator.
  process.exit(auth && wsConnect && subAck && nonDepth && leaks === 0 ? 0 : 1);
}, duration * 1000);
```
